<form action="https://localhost:8080/login_anne" method="post" encType="multipart/form-data" className="login_main_form"> 이런식으로 react tsx를 작성하면, 스프링 부트로 form 값을 보낼 수 있나?    +2                    아니오, 작성하신 JSX 형태 그대로는 스프링 부트로 정상적인 전송이 되지 않습니다. 리액트는 기본 HTML 폼 제출(페이지 새로고침) 방식 대신, 자바스크립트의 FormData 객체와 Axios 또는 fetch 같은 비동기 통신을 사용해야 합니다.올바른 구현 방법이벤트 방지: onSubmit에서 e.preventDefault()로 기본 동작을 막아야 합니다.FormData 사용: 입력값을 FormData 객체에 append하여 담아줍니다.비동기 전송: axios.post나 fetch를 이용해 백엔드로 보냅니다.리액트 (TSX) 예시 코드tsxconst handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
  e.preventDefault(); // 페이지 새로고침 방지
  
  const formData = new FormData();
  formData.append("username", "anne");
  // formData.append("file", fileObject); // 파일이 있다면 추가

  try {
    const response = await axios.post("https://localhost:8080/login_anne", formData, {
      headers: {
        "Content-Type": "multipart/form-data",
      },
    });
    console.log(response.data);
  } catch (error) {
    console.error(error);
  }
};

// 렌더링 부분
<form onSubmit={handleSubmit} className="login_main_form">
  {/* 입력 필드들 */}
  <button type="submit">전송</button>
</form>
