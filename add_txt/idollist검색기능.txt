
import { useState } from 'react';

function App() {
  // 1. input의 입력값을 관리할 상태(state) 생성
  const [searchTerm, setSearchTerm] = useState<string>('');

  // 2. 상자 데이터를 배열로 정의 (id는 숫자로 구성하되 className은 문자열로 변환)
  const boxes = Array.from({ length: 12 }, (_, i) => ({
    id: i + 1,
    className: String(i + 1),
    text: `${i + 1}번상자`
  }));

  // 3. 입력값(searchTerm)이 포함된 className을 가진 상자만 필터링
  const filteredBoxes = boxes.filter((box) =>
    box.className.includes(searchTerm)
  );

  return (
    <>
      <div className="full">
        {/* 4. input의 value와 onChange 이벤트를 연결 */}
        <input
          type="text"
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
          placeholder="숫자를 검색하세요 (예: 1, 2)"
        />

        {/* 5. 필터링된 배열만 화면에 출력 */}
        {filteredBoxes.map((box) => (
          <div key={box.id} className={box.className}>
            {box.text}
          </div>
        ))}
      </div>
    </>
  );
}

export default App;

## 💡 수정된 React.tsx 코드
아래 코드를 그대로 복사해서 사용하시면 됩니다. 이미지 파일은 프로젝트의 public/images/ 폴더 등에 가수의 영어 이름이나 ID에 맞춰 저장해 두시면 매칭됩니다.

import { useState } from 'react';
// 가수의 데이터 구조를 정의하는 타입 인터페이스interface Artist {
  id: number;
  koName: string;
  enName: string;
  imgUrl: string;
}
function App() {
  const [searchTerm, setSearchTerm] = useState<string>('');

  // 1. 한국어 이름, 영어 이름, 이미지를 포함한 아티스트 데이터 배열
  const artists: Artist[] = [
    { id: 1, koName: '트리플에스', enName: 'triples', imgUrl: '/images/triples.jpg' },
    { id: 2, koName: '투모로우바이투게더', enName: 'tomorrow x together', imgUrl: '/images/txt.jpg' },
    { id: 3, koName: '악뮤', enName: 'akmu', imgUrl: '/images/akmu.jpg' },
    { id: 4, koName: '아이오아이', enName: 'ioi', imgUrl: '/images/ioi.jpg' },
    { id: 5, koName: '스트레이 키즈', enName: 'stray kids', imgUrl: '/images/stray_kids.jpg' },
    { id: 6, koName: '아일릿', enName: 'illit', imgUrl: '/images/illit.jpg' },
    { id: 7, koName: '투어스', enName: 'tws', imgUrl: '/images/tws.jpg' },
    { id: 8, koName: '리센느', enName: 'rescene', imgUrl: '/images/rescene.jpg' },
    { id: 9, koName: '10센치', enName: '10cm', imgUrl: '/images/10cm.jpg' },
    { id: 10, koName: '에스파', enName: 'aespa', imgUrl: '/images/aespa.jpg' },
    { id: 11, koName: '올데이 프로젝트', enName: 'allday project', imgUrl: '/images/allday_project.jpg' },
    { id: 12, koName: '데이식스', enName: 'day6', imgUrl: '/images/day6.jpg' },
  ];

  // 2. 한국어 또는 영어 이름에 검색어가 포함되어 있는지 필터링 (대소문자 구분 없음)
  const filteredArtists = artists.filter((artist) => {
    const sTerm = searchTerm.toLowerCase(); // 검색어를 소문자로 통일
    const koMatch = artist.koName.includes(sTerm);
    const enMatch = artist.enName.toLowerCase().includes(sTerm); // 영어 이름도 소문자로 변환 후 비교
    
    return koMatch || enMatch; // 둘 중 하나라도 맞으면 포함
  });

  return (
    <>
      <div className="full">
        {/* 검색 입력창 */}
        <input
          type="text"
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
          placeholder="가수 이름 검색 (한글 또는 영어)"
          style={{ padding: '10px', width: '300px', marginBottom: '20px' }}
        />

        {/* 필터링된 아티스트 목록 출력 */}
        <div className="artist-list" style={{ display: 'flex', flexWrap: 'wrap', gap: '20px' }}>
          {filteredArtists.map((artist) => (
            <div key={artist.id} className={artist.enName.replace(/\s+/g, '-')}>
              {/* 이미지 출력 */}
              <img 
                src={artist.imgUrl} 
                alt={artist.koName} 
                style={{ width: '150px', height: '150px', objectFit: 'cover', borderRadius: '8px' }} 
              />
              {/* h1 태그 안에 이름 출력 */}
              <h1>{artist.koName} ({artist.enName})</h1>
            </div>
          ))}
        </div>
      </div>
    </>
  );
}
export default App;

## 🛠️ 핵심 변경 및 개선 사항

   1. 다중 조건 필터링 (|| 연산자)
   * koMatch || enMatch를 사용하여 입력한 값이 한국어 이름에 포함되거나 영어 이름에 포함되면 모두 결과에 나타나도록 만들었습니다.
   2. 영어 대소문자 무시 (toLowerCase())
   * 사용자가 대문자(AESPA)나 소문자(aespa) 중 어떻게 검색하더라도 모두 매칭되도록 검색어와 영어 이름을 모두 소문자로 변환한 뒤 비교합니다.
   3. h1 및 img 태그 동적 바인딩
   * 각 데이터에 맞게 <h1>{artist.koName}</h1> 형태로 텍스트를 출력합니다.
      * <img> 태그의 src 속성에 각 가수의 이미지 경로(artist.imgUrl)가 동적으로 들어가도록 구현했습니다.
   4. 공백 제거 클래스네임 (replace)
   * div의 className으로 영어 이름을 사용할 때, tomorrow x together처럼 공백이 있으면 클래스가 여러 개로 쪼개집니다. 이를 방지하기 위해 replace(/\s+/g, '-')를 사용해 tomorrow-x-together 형태로 클래스명이 등록되도록 처리했습니다.
