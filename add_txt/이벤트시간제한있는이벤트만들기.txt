시간 제한


import { useEffect } from 'react';
import { useNavigate } from 'react-router-dom';

export default function TimeoutRedirectPage() {
  const navigate = useNavigate();
  const TIMEOUT_MS = 5000; // 5초 설정

  useEffect(() => {
    // 1. 타이머 설정
    const timer = setTimeout(() => {
      navigate(-1); // -1은 이전 페이지로 이동을 뜻합니다.
    }, TIMEOUT_MS);

    // 2. 클린업 함수 (중요!)
    return () => clearTimeout(timer);
  }, [navigate]);

  return (
    <div style={{ padding: '20px', textAlign: 'center' }}>
      <h1>시간 초과 페이지</h1>
      <p>5초 후에 이전 페이지로 자동으로 돌아갑니다.</p>
    </div>
  );
}



네, setTimeout을 사용하는 것이 정확한 방법입니다.
React(TypeScript)에서는 컴포넌트가 처음 렌더링될 때 useEffect 훅 내부에서 setTimeout을 실행하고, 페이지를 이동시키면 됩니다.

## ⚠️ 주의해야 할 핵심 포인트

* 클린업 함수(clearTimeout) 필수: 사용자가 5초가 지나기 전에 스스로 '뒤로 가기'를 누르거나 다른 링크로 이동할 수 있습니다. 이때 타이머를 끄지 않으면, 다른 페이지에 가 있는 상태에서 갑자기 navigate(-1)이 실행되어 버리는 버그가 발생합니다. return () => clearTimeout(timer); 코드가 이 버그를 막아줍니다.
* 이전 페이지 이동: navigate(-1)을 사용하면 브라우저 히스토리 상 바로 직전 페이지로 이동합니다. 만약 특정 경로(예: 홈 화면)로 보내고 싶다면 navigate('/')와 같이 주소를 직접 입력하면 됩니다.

import React, { useState, useEffect, useRef } from 'react';
import { useNavigate } from 'react-router-dom';

export default function TimeoutForm() {
  const navigate = useNavigate();
  
  // 상태 관리
  const [inputValue, setInputValue] = useState<string>('');
  const [timeLeft, setTimeLeft] = useState<number>(10);
  
  // 설정 정보
  const CORRECT_ANSWER = "정답"; // 검증할 정확한 값
  const timerRef = useRef<NodeJS.Timeout | null>(null);

  // 1단계: 10초 카운트다운 및 시간 초과 시 페이지 이동
  useEffect(() => {
    timerRef.current = setInterval(() => {
      setTimeLeft((prev) => {
        if (prev <= 1) {
          clearInterval(timerRef.current!);
          navigate('/timeout-page'); // 시간 초과 시 이동할 라우트
          return 0;
        }
        return prev - 1;
      });
    }, 1000);

    // 컴포넌트 언마운트 시 타이머 클리어 (메모리 누수 방지)
    return () => {
      if (timerRef.current) clearInterval(timerRef.current);
    };
  }, [navigate]);

  // 2단계: 양식 제출 핸들러
  const handleSubmit = (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();

    if (inputValue === CORRECT_ANSWER) {
      if (timerRef.current) clearInterval(timerRef.current); // 타이머 멈춤
      alert("성공했습니다!");
      navigate('/success-page'); // 성공 시 이동할 라우트
    } else {
      alert("올바른 값이 아닙니다. 다시 시도하세요!");
    }
  };

  return (
    <div style={{ padding: '20px', textAlign: 'center' }}>
      <h2>제한 시간 내에 입력하세요!</h2>
      <h3 style={{ color: timeLeft <= 3 ? 'red' : 'black' }}>
        남은 시간: {timeLeft}초
      </h3>
      
      <form onSubmit={handleSubmit}>
        <input
          type="text"
          value={inputValue}
          onChange={(e) => setInputValue(e.target.value)}
          placeholder={`"${CORRECT_ANSWER}"를 입력하세요`}
          style={{ padding: '8px', marginRight: '8px' }}
        />
        <button type="submit" style={{ padding: '8px 16px' }}>
          제출
        </button>
      </form>
    </div>
  );
}
