import { useState } from "react";

export default function IdnttEvent() {
    // 현재 화면에 보여줄 단계 (0: 1번만, 1: 2번까지, 2: 3번까지)
    const [step, setStep] = useState<number>(0);

    return (
        <div>
            <section>
                {/* 상시 노출 */}
                <p onClick={() => setStep(1)} style={{ cursor: "pointer" }}>1번</p>
                
                {/* step이 1 이상일 때 2번 노출 */}
                {step >= 1 && <p onClick={() => setStep(2)} style={{ cursor: "pointer" }}>2번</p>}
                
                {/* step이 2 이상일 때 3번 노출 */}
                {step >= 2 && <p>3번</p>}
            </section>
        </div>
    );
}
