import { useState } from "react";

interface IdnttEventNodeType {
    Idntt_e_mem_name: string;
    Idntt_e_link: IdnttEventNodeType | null;
}

export default function IdnttEvent() {
    // 1번 노드 내부에 2번, 3번 구조를 단일 상태로 관리합니다.
    const [Idntt_e_rootNode, setIdntt_e_RootNode] = useState<IdnttEventNodeType>({
        Idntt_e_mem_name: "1번",
        Idntt_e_link: null,
    });

    // 1번 클릭 시 -> 1번에 2번을 연결
    const Idntt_e_connectNode2 = () => {
        setIdntt_e_RootNode({
            ...Idntt_e_rootNode,
            Idntt_e_link: { Idntt_e_mem_name: "2번", Idntt_e_link: null }
        });
    };

    // 2번 클릭 시 -> 1번 내부의 2번에 3번을 연결
    const Idntt_e_connectNode3 = () => {
        if (!Idntt_e_rootNode.Idntt_e_link) return;

        setIdntt_e_RootNode({
            ...Idntt_e_rootNode,
            Idntt_e_link: {
                ...Idntt_e_rootNode.Idntt_e_link,
                Idntt_e_link: { Idntt_e_mem_name: "3번", Idntt_e_link: null }
            }
        });
    };

    // 3번 클릭 시 -> 4번을 연결
    const Idntt_e_connectNode4 = () => {
        if (!Idntt_e_rootNode.Idntt_e_link) return;

        setIdntt_e_RootNode({
            ...Idntt_e_rootNode,
            Idntt_e_link: {
                ...Idntt_e_rootNode.Idntt_e_link,
                Idntt_e_link: { Idntt_e_mem_name: "3번", Idntt_e_link: null }
            }
        });
    };

    // 참조 변수를 쉽게 쓰기 위해 껍데기 변수 생성
    const Idntt_e_node2 = Idntt_e_rootNode.Idntt_e_link;
    const Idntt_e_node3 = Idntt_e_node2?.Idntt_e_link;

    return (
        <div>
            <section>
                {/* 1번 노드 */}
                <p onClick={Idntt_e_connectNode2} style={{ cursor: "pointer" }}>
                    {Idntt_e_rootNode.Idntt_e_mem_name}
                </p>

                {/* 2번 노드: 1번의 링크가 존재할 때 노출 */}
                {Idntt_e_node2 !== null && (
                    <p onClick={Idntt_e_connectNode3} style={{ cursor: "pointer" }}>
                        {Idntt_e_node2.Idntt_e_mem_name}
                    </p>
                )}

                {/* 3번 노드: 2번의 링크가 존재할 때 노출 */}
                {Idntt_e_node3 != null && (
                    <p>{Idntt_e_node3.Idntt_e_mem_name}</p>
                )}

                {/* 4번 노드: 2번의 링크가 존재할 때 노출 */}
                {Idntt_e_node4 != null && (
                    <p>{Idntt_e_node4.Idntt_e_mem_name}</p>
                )}
            </section>
        </div>
    );
}
